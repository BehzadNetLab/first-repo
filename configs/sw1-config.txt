! Router-on-a-Stick - SW1 Configuration

enable
configure terminal

vlan 10
 name STAFF-10

vlan 20
 name ADMIN-20

interface GigabitEthernet0/1
 switchport mode trunk
 switchport trunk allowed vlan 10,20
 switchport nonegotiate
 no shutdown

interface range FastEthernet0/3-4
 switchport mode trunk
 switchport trunk allowed vlan 10,20
 switchport nonegotiate
 no shutdown

end
